# Installed inside the SageMaker training container (source_code.requirements) and used locally.
#
# torch is deliberately NOT pinned: the SageMaker PyTorch 2.10 container (py313, CUDA 13.0)
# ships its own CUDA build, and pinning a different version here would make every job
# reinstall torch over it. Locally, the `gist` env has torch 2.14.0 installed directly.
#
# transformers 5.x renamed two things this code depends on: `torch_dtype` -> `dtype` on
# from_pretrained, and TrainingArguments dropped `warmup_ratio` for `warmup_steps`.
# Do not relax the transformers pin below 5.x without reverting those.
transformers==5.17.0
peft==0.21.0
accelerate==1.15.0
wandb==0.30.0
pyyaml==6.0.3
numpy==2.5.3

# CUDA-only 4-bit quantization for QLoRA. First verified by the SageMaker smoke job.
bitsandbytes==0.50.2 ; platform_system == "Linux"

# Local QLoRA on Apple silicon (train_mlx.py). Never installed in the Linux container.
mlx==0.32.2 ; platform_system == "Darwin"
mlx-lm==0.31.3 ; platform_system == "Darwin"

# The launcher (launch_sagemaker.py) runs on your laptop only and needs sagemaker==3.22.1 and
# botocore[crt] (the `aws login` credential provider requires it), installed separately - they
# have no place inside the training container.
